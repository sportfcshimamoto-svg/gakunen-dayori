/**
 * 週案（Googleスプレッドシート）に「週のタブを作る」メニューを足すプログラム。
 * 使い方：週案で「拡張機能 → Apps Script」を開き、このファイルの中身をすべて貼って保存する。
 * 週案を開きなおすと、上のメニューに「週案」が出る。
 *
 * やること：時間割（組ごと。2クラスでも3クラスでも、週案にある組の数だけ）と期間を選ぶと、その期間の週ごとに、
 * 同じA週/B週のいちばん新しい週のタブをコピーして新しいタブを作る。
 * 月〜金のマス（朝学〜宿題）は空にして、日付・A週/B週・時間割の教科だけ入れる。
 * 週案にもうある週は作らない。タブの名前は「12月14日」の形。
 * 時間割や週案をほかの場所に送ることはない（このスプレッドシートの中だけで動く）。
 */

var DAYS = ['月', '火', '水', '木', '金'];
var SLOTS = ['朝', '1', '2', '3', '4', '5', '6'];
var SHORT = { 国: '国語', 社: '社会', 算: '算数', 理: '理科', 生: '生活', 音: '音楽', 図: '図工', 体: '体育', 家: '家庭', 道: '道徳', 特: '特活', 総: '総合', 外: '外国語' };

// ---------- メニュー・画面 ----------
function onOpen() {
  SpreadsheetApp.getUi().createMenu('週案').addItem('週のタブを作る', 'showSidebar')
    .addSeparator().addItem('5年生の形にそろえる（宿題・音・連の行）', 'addOtoRenRows').addToUi();
}

// ---------- 5年生の形にそろえる：「○組」の見出し、宿題の行、その下の「音」「連」の行（連絡帳の「音」「連」に書く） ----------
var EXTRA_ROWS = ['音', '連'];

/** メニュー：週案を5年生の形にそろえる。
 *  すべての表のタブ（過去の週も）：①月曜の左のマスが空なら「1組」「2組」…を入れる ②タブの名前の「～」をとって「8月25日」の形に
 *  原本（「A週」「後期」などのひな形）と今週から先の週：③6校時の下に「宿題」の行がなければ足す ④宿題の下に「音」「連」の行を足す
 *  （過去の週は行を足さない。①②は時数表アプリが1年分の週を読めるようにするため） */
function addOtoRenRows() {
  var ui = SpreadsheetApp.getUi();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var now = new Date();
  var thisMon = mondayOf_(iso_(new Date(Date.UTC(now.getFullYear(), now.getMonth(), now.getDate()))));
  var future = {};
  existingWeeks_().forEach(function (w) { if (w.monday >= thisMon) future[w.name] = 1; });
  var tables = ss.getSheets().filter(function (s) {
    var vals = displayValues_(s, 6);
    return Object.keys(shapeOf_(vals).classes).length > 0 || bareBlocks_(vals).length > 0;   // 時間割の表のあるタブ
  });
  var rowTargets = tables.filter(function (s) {
    var name = s.getName();
    if (!isWeekName_(name)) return !/^\s*[0-9０-９]/.test(name);   // ひな形（原本・A週など）。「97実習生用」のような週の写しは足さない
    if (future[name]) return true;
    // 見出しのない週案は、タブの名前で今週から先か決める（決めきれない名前は、どの読み方でも先のときだけ）
    if (Object.keys(shapeOf_(displayValues_(s, 6)).classes).length) return false;
    var c = mondayCandidates_(name, 0, schoolYear_());
    return c.length > 0 && c.every(function (m) { return m >= thisMon; });
  });
  var renames = tables.filter(function (s) { return isWeekName_(s.getName()) && needsRename_(s.getName()); });
  if (!tables.length) { ui.alert('時間割の表（月〜金の見出し）があるタブが見つかりませんでした。'); return; }
  var ok = ui.alert('5年生の形にそろえる',
    '週案を5年生の週案の形にそろえます（もうそろっているところは、そのまま）。\n\n' +
    '■ すべての週のタブ（' + tables.length + 'タブ）\n' +
    '・月曜の左のマスが空なら「1組」「2組」…を入れる\n' +
    (renames.length ? '・タブの名前の「～」をとって「8月25日」の形にする（' + renames.length + 'タブ）\n' : '') +
    '\n■ 原本と、今週から先の週のタブ\n' +
    '・6校時の下に「宿題」の行、その下に「音」「連」の行を足す\n' +
    rowTargets.map(function (s) { return s.getName(); }).join('、') +
    '\n\nほかの行やマスはそのままです。これから「週のタブを作る」で作るタブにも、この行が入ります。よろしいですか？', ui.ButtonSet.OK_CANCEL);
  if (ok !== ui.Button.OK) return;
  var heads = 0, rows = [], notes = [];
  tables.forEach(function (s) {
    var a = addClassHeads_(s);
    if (a === null) notes.push(s.getName() + '：月曜の左のマスに文字があるので、組の見出しを入れられませんでした');
    else if (a) heads++;
  });
  rowTargets.forEach(function (s) {
    if (!Object.keys(shapeOf_(displayValues_(s, 6)).classes).length) return;
    var b = addHomeworkRow_(s), c = addExtraRows_(s);
    if (b || c) rows.push(s.getName());
  });
  var renamed = renameWeekTabs_(ss, notes);
  SpreadsheetApp.flush();
  var msg = [];
  if (heads) msg.push('組の見出しを入れました：' + heads + 'タブ');
  if (renamed.length) msg.push('タブの名前を直しました：' + renamed.join('、'));
  if (rows.length) msg.push('「宿題」「音」「連」の行を足しました：' + rows.join('、'));
  ui.alert((msg.length ? msg.join('\n\n') : 'どのタブも、もう5年生の形でした。') + (notes.length ? '\n\n' + notes.join('\n') : ''));
}

/** 名前を直すタブ：後ろに「～」がある、または全角の数字・「／」など（5年生は「1130」「8月25日」の形） */
function needsRename_(name) { return /[~〜～]\s*$/.test(name) || norm_(name) !== String(name).trim(); }

/** 週のタブの名前を「8月25日」の形にする（「825~」→「8月25日」、「12／7～」→「12月7日」）。直した「前→後」の一覧 */
function renameWeekTabs_(ss, notes) {
  var weeks = existingWeeks_(), out = [];
  weeks.forEach(function (w) {
    if (!needsRename_(w.name)) return;
    var sh = ss.getSheetByName(w.name);
    if (!sh) return;
    var md = nameDate_(w.name, w.monday);
    if (!md) { notes.push(w.name + '：名前から日付が決められないので、名前は直していません'); return; }
    var to = md.m + '月' + md.d + '日';
    if (ss.getSheetByName(to)) { notes.push(w.name + '：「' + to + '」というタブがもうあるので、名前は直していません'); return; }
    sh.setName(to);
    out.push(w.name + '→' + to);
  });
  return out;
}

/** タブの名前が表す日（月曜とは限らない。祝日の週は火曜の名前のことも）。数字だけの名前は、月曜から4日のうちに入る読み方 */
function nameDate_(name, monday) {
  var k = weekKey_(name);
  var sl = /^(\d{1,2})(?:\/|月)(\d{1,2})日?$/.exec(k);
  if (sl) return { m: Number(sl[1]), d: Number(sl[2]) };
  if (!/^\d{2,4}$/.test(k)) return null;
  var mon = dateOf_(monday);
  for (var n = 1; n <= 2; n++) {
    var m = Number(k.slice(0, n)), d = Number(k.slice(n));
    if (k.length - n < 1 || k.length - n > 2 || m < 1 || m > 12 || d < 1 || d > 31) continue;
    for (var i = 0; i < 5; i++) {
      var day = new Date(mon.getTime() + i * 86400000);
      if (day.getUTCMonth() + 1 === m && day.getUTCDate() === d) return { m: m, d: d };
    }
  }
  return null;
}

/** 「○組」の見出しがない表：上の4行で、月〜金が横に並んでいるところ（左どなりのマスが見出しの場所）。左から順 */
function bareBlocks_(vals) {
  var out = [];
  for (var r = 1; r <= Math.min(vals.length, 4); r++) {
    var maxC = vals[r - 1].length;
    for (var c = 2; c + 4 <= maxC; c++) {
      if (DAYS.every(function (d, i) { return norm_(at_(vals, c + i, r)) === d; })) out.push({ r: r, labelCol: c - 1, label: norm_(at_(vals, c - 1, r)) });
    }
  }
  return out.sort(function (a, b) { return a.labelCol - b.labelCol; });
}

/** 「○組」の見出しがなければ、月曜の左のマスに左から「1組」「2組」…を入れる。入れたら true、もうあれば false、入れられなければ null */
function addClassHeads_(sh) {
  var vals = displayValues_(sh);
  if (Object.keys(shapeOf_(vals).classes).length) return false;
  var blocks = bareBlocks_(vals);
  if (!blocks.length || blocks.some(function (b) { return b.label && !/組$/.test(b.label); })) return null;
  blocks.forEach(function (b, i) { sh.getRange(b.r, b.labelCol).setValue((i + 1) + '組'); });
  return true;
}

/** 6校時（3行）の下に「宿題」の行がなければ足す（書式は上の行から。マスは空）。足したら true */
function addHomeworkRow_(sh) {
  var shape = shapeOf_(displayValues_(sh));
  var names = Object.keys(shape.classes);
  if (!names.length || names.every(function (c) { return shape.classes[c].rows['宿題']; })) return false;
  // 6校時のかたまり（または、6校時のところに書いた「宿題」）のいちばん下の行の、次に足す
  var after = 0;
  names.forEach(function (c) {
    var cl = shape.classes[c];
    if (cl.rows['6']) after = Math.max(after, blockEnd_(sh, cl.rows['6'], cl.labelCol));
    if (cl.rows['宿題']) after = Math.max(after, blockEnd_(sh, cl.rows['宿題'], cl.labelCol));
  });
  if (!after) return false;
  var row = insertPlainRow_(sh, after);
  names.forEach(function (c) {
    var cl = shape.classes[c];
    if (!cl.rows['宿題']) sh.getRange(row, cl.labelCol).setValue('宿題');
  });
  return true;
}

/** 見出しのマス（結合していれば、そのかたまり）のいちばん下の行 */
function blockEnd_(sh, row, col) {
  var m = sh.getRange(row, col).getMergedRanges();
  return m && m.length ? m[0].getLastRow() : row;
}

/** after の行の下に、空の1行を足す（書式は上の行から。結合はしない。月〜金の色はなし）。足した行の番号 */
function insertPlainRow_(sh, after) {
  sh.insertRowsAfter(after, 1);
  var row = after + 1;
  var all = sh.getRange(row, 1, 1, sh.getMaxColumns());
  all.breakApart();
  all.clearContent();
  all.setBackground(null);
  sh.setRowHeight(row, Math.min(sh.getRowHeight(after), 40));
  return row;
}

/** 週案のどこかのタブ（ひな形）に「音」「連」の行があるか */
function hasExtraRows_(ss) {
  return ss.getSheets().some(function (s) {
    if (isWeekName_(s.getName())) return false;
    var cls = shapeOf_(displayValues_(s)).classes;
    return Object.keys(cls).some(function (c) { return EXTRA_ROWS.every(function (l) { return cls[c].rows[l]; }); });
  });
}

/** 1つのタブの「宿題」の行の下に、ない「音」「連」の行を足す（書式は宿題の行と同じ、マスは空）。足したら true */
function addExtraRows_(sh) {
  var added = false;
  EXTRA_ROWS.forEach(function (lab) {
    var shape = shapeOf_(displayValues_(sh));
    var names = Object.keys(shape.classes).filter(function (c) { return shape.classes[c].rows['宿題']; });
    if (!names.length || names.every(function (c) { return shape.classes[c].rows[lab]; })) return;
    // 宿題の行（結合していればそのかたまり）と、前に足した行のいちばん下の、次に足す
    var after = 0;
    names.forEach(function (c) {
      var cl = shape.classes[c];
      after = Math.max(after, blockEnd_(sh, cl.rows['宿題'], cl.labelCol));
      EXTRA_ROWS.slice(0, EXTRA_ROWS.indexOf(lab)).forEach(function (l) { if (cl.rows[l]) after = Math.max(after, cl.rows[l]); });
    });
    var row = insertPlainRow_(sh, after);
    names.forEach(function (c) { if (!shape.classes[c].rows[lab]) sh.getRange(row, shape.classes[c].labelCol).setValue(lab); });
    added = true;
  });
  return added;
}

/** 画面を開いたときに渡すもの：覚えている時間割、週案にある週 */
function getState() {
  var props = PropertiesService.getDocumentProperties();
  var weeks = existingWeeks_();
  var last = weeks[weeks.length - 1];
  var classes = classNames_(last && last.name), text = {};
  classes.forEach(function (c) { text[c] = props.getProperty('tt_' + c) || ''; });
  return {
    classes: classes,
    text: text,
    weeks: weeks.map(function (w) { return { name: w.name, monday: w.monday, ab: w.ab }; }),
    from: last ? addDays_(last.monday, 7) : '',
    bunkatsu: props.getProperty('bunkatsu') !== '0',   // 算数の下に「分割」を入れるか（はじめは入れる）
  };
}

/** 時間割の文字を読んで、表（確かめ用）を返す。文字はこのスプレッドシートに覚える */
function checkTimetable(cls, text) {
  PropertiesService.getDocumentProperties().setProperty('tt_' + cls, String(text || ''));
  if (!String(text || '').trim()) return null;
  return parseTimetable_(parseTsv_(text));
}

/** 算数の下に「分割」を入れるかを、このスプレッドシートに覚える */
function setBunkatsu(on) {
  PropertiesService.getDocumentProperties().setProperty('bunkatsu', on ? '1' : '0');
}

/** 作る週の一覧（画面に出す） */
function previewWeeks(from, to, abOverride) {
  return planWeeks_(from, to, existingWeeks_(schoolYearOf_(from)), abOverride || {});
}

/** 週のタブを作る */
function makeWeeks(from, to, abOverride, bunkatsu) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var props = PropertiesService.getDocumentProperties();
  if (bunkatsu === undefined) bunkatsu = props.getProperty('bunkatsu') !== '0';
  var grids = {};
  classNames_().forEach(function (c) {
    var t = props.getProperty('tt_' + c);
    if (!t) return;
    var p = parseTimetable_(parseTsv_(t));
    if (p.ok) grids[c] = p.grid;
  });
  if (!Object.keys(grids).length) throw new Error('時間割がまだ読み込まれていません');
  var existing = existingWeeks_(schoolYearOf_(from));
  var list = planWeeks_(from, to, existing, abOverride || {}).filter(function (w) { return !w.exists; });
  if (!list.length) throw new Error('作る週がありません。期間を確かめてください');
  var made = [], notes = [];
  list.forEach(function (w) {
    var tpl = pickTemplate_(ss, existing, w.ab);
    if (!tpl) throw new Error(w.ab + '週のひな形になるタブが見つかりません');
    if (ss.getSheetByName(w.name)) { notes.push('「' + w.name + '」という名前のタブがもうあるので作りませんでした'); return; }
    var sh = tpl.copyTo(ss).setName(w.name);
    ss.setActiveSheet(sh);
    ss.moveActiveSheet(ss.getNumSheets());
    if (hasExtraRows_(ss)) { addHomeworkRow_(sh); addExtraRows_(sh); }   // 原本に「音」「連」の行があれば、新しいタブにも
    notes = notes.concat(fillWeek_(sh, w, grids, colorSource_(ss, w.ab), bunkatsu));
    existing.push({ name: w.name, monday: w.monday, ab: w.ab });   // 次の週のひな形は、いま作ったタブ
    made.push(w.name);
  });
  SpreadsheetApp.flush();
  return { made: made, notes: notes };
}

// ---------- 週案のタブを読む ----------
/** 週案にある組（「1組」「2組」「3組」…）。いちばん新しい週のタブを見る。なければ、組の見出しのあるタブを全部見る */
function classNames_(lastName) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var names = {};
  var look = function (sh) { Object.keys(shapeOf_(displayValues_(sh, 4)).classes).forEach(function (c) { names[c] = 1; }); };
  if (!lastName) { var w = existingWeeks_(); if (w.length) lastName = w[w.length - 1].name; }
  var sh = lastName && ss.getSheetByName(lastName);
  if (sh) look(sh);
  if (!Object.keys(names).length) ss.getSheets().forEach(look);
  var list = Object.keys(names).sort(function (a, b) { return parseInt(a, 10) - parseInt(b, 10); });
  return list.length ? list : ['1組', '2組'];
}

function displayValues_(sh, maxRows) {
  var r = sh.getLastRow(), c = sh.getLastColumn();
  if (maxRows) r = Math.min(r, maxRows);
  if (!r || !c) return [];
  return sh.getRange(1, 1, r, c).getDisplayValues();
}

/** 週案にある週のタブ：[{ name, monday, ab, sheet }]（月曜の順） */
/** sy：タブの名前（例「4月6日」）を何年度として読むか。期間を選んだら、その最初の日の年度（3月に次の年度のタブを作るときのため） */
function existingWeeks_(sy) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  sy = sy || schoolYear_();
  var found = [];
  ss.getSheets().forEach(function (sh) {
    var name = sh.getName();
    if (!isWeekName_(name)) return;   // 日付のタブだけ見る（ひな形は見ない）
    var info = weekInfo_(displayValues_(sh, 4));   // 見出し（上の4行）だけ読めばよい
    if (!info) return;
    found.push({ name: name, cands: mondayCandidates_(name, info.mondayDay, sy), ab: info.ab });
  });
  return resolveMondays_(found);
}

/** タブの中身から「○組」と月〜金の見出し、A週/B週、月曜の日を見つける */
function weekInfo_(vals) {
  var shape = shapeOf_(vals);
  var names = Object.keys(shape.classes);
  if (!names.length) return null;
  var mondayDay = 0;
  names.forEach(function (n) {
    var cl = shape.classes[n];
    var dr = dateRowOf_(vals, cl);
    if (dr && !mondayDay) {
      var m = /(\d{1,2})日?$/.exec(norm_(at_(vals, cl.dayCols['月'], dr.row)));
      if (m) mondayDay = Number(m[1]);
    }
  });
  return { ab: shape.ab, mondayDay: mondayDay };
}

/** 「1組」「2組」のかたまり、A週/B週のマス、日付の行、校時の行。行・列は1から数える */
function shapeOf_(vals) {
  var maxR = vals.length, maxC = maxR ? vals[0].length : 0;
  var classes = {}, abCell = null, ab = null;
  for (var r = 1; r <= Math.min(maxR, 4); r++) for (var c = 1; c <= maxC; c++) {
    var v = norm_(at_(vals, c, r));
    if (/^\d+組$/.test(v)) {
      var dayCols = {};
      for (var k = 1; k <= 7; k++) { var dv = norm_(at_(vals, c + k, r)); if (DAYS.indexOf(dv) >= 0) dayCols[dv] = c + k; }
      // 見出しの1マスにメモなどが書かれていても、となりから月〜金が順に並んでいれば、その並びで読む
      if (Object.keys(dayCols).length < 5 && DAYS.filter(function (d, i) { return norm_(at_(vals, c + 1 + i, r)) === d; }).length >= 4) {
        DAYS.forEach(function (d, i) { dayCols[d] = c + 1 + i; });
      }
      if (Object.keys(dayCols).length === 5) classes[v] = { labelCol: c, headRow: r, dayCols: dayCols, rows: {} };
    }
    if (!abCell && /^[AB]週$/.test(v)) { abCell = { r: r, c: c }; ab = v.charAt(0); }
  }
  Object.keys(classes).forEach(function (n) {
    var cl = classes[n];
    for (var r = cl.headRow + 1; r <= maxR; r++) {
      var lab = norm_(at_(vals, cl.labelCol, r));
      if (/^(朝学|朝学習|モジュール)$/.test(lab)) cl.rows['朝'] = r;
      else if (/^[1-6](\.0+)?$/.test(lab)) cl.rows[String(parseInt(lab, 10))] = r;
      else if (lab === '宿題') cl.rows['宿題'] = r;
      else if (EXTRA_ROWS.indexOf(lab) >= 0 && cl.rows['宿題'] && !cl.rows[lab]) cl.rows[lab] = r;   // 宿題の下の「音」「連」
    }
    cl.firstRow = cl.rows['朝'] || cl.rows['1'] || 0;
    cl.lastRow = Math.max(cl.rows['宿題'] || 0, cl.rows['6'] ? cl.rows['6'] + 2 : 0);
    EXTRA_ROWS.forEach(function (lab) { if (cl.rows[lab]) cl.lastRow = Math.max(cl.lastRow, cl.rows[lab]); });
  });
  return { classes: classes, abCell: abCell, ab: ab };
}

/** 日付の行：曜日の見出しの上の行の「14日」「1月4日」、または見出しの下の行の「12/14」「12月14日」（どちらもなければ、上に行があれば上） */
function dateRowOf_(vals, cl) {
  var col = cl.dayCols['月'];
  var above = cl.headRow > 1 ? norm_(at_(vals, col, cl.headRow - 1)) : null;
  var below = norm_(at_(vals, col, cl.headRow + 1));
  if (above && /^(\d{1,2}月)?\d{1,2}日?$/.test(above)) return { row: cl.headRow - 1, full: false };
  if (/^(\d{4}[\/\-年])?\d{1,2}[\/\-月]\d{1,2}日?(\(.\))?$/.test(below) || /^\d{1,2}日$/.test(below)) return { row: cl.headRow + 1, full: true };
  if (cl.headRow > 1) return { row: cl.headRow - 1, full: false };
  return null;
}

/** タブの名前（「1214」「12/14」「12月14日」）と月曜の日から、月曜日の日付の候補（祝日の週は火曜の名前のこともある） */
function mondayCandidates_(name, mondayDay, schoolYear) {
  var digits = weekKey_(name);
  var sl =/^(\d{1,2})(?:\/|月)(\d{1,2})日?$/.exec(digits);
  if (sl) digits = sl[1] + ('0' + sl[2]).slice(-2);
  if (!/^\d{2,4}$/.test(digits)) return [];
  var out = [], byName = [];
  var ks = sl ? [sl[1].length] : [1, 2];
  ks.forEach(function (k) {
    var m = Number(digits.slice(0, k)), d = Number(digits.slice(k));
    if (digits.length - k < 1 || digits.length - k > 2 || m < 1 || m > 12 || d < 1 || d > 31) return;
    var year = m >= 4 ? schoolYear : schoolYear + 1;
    var named = new Date(Date.UTC(year, m - 1, d));
    if (named.getUTCMonth() !== m - 1) return;
    for (var back = 0; back <= 4; back++) {
      var mon = new Date(named.getTime() - back * 86400000);
      if (mon.getUTCDay() !== 1) continue;
      if (byName.indexOf(iso_(mon)) < 0) byName.push(iso_(mon));
      if (mon.getUTCDate() === mondayDay && out.indexOf(iso_(mon)) < 0) out.push(iso_(mon));
    }
  });
  // 週案の中の日付が名前と合えばそれ。日付がない・ひな形のまま直っていないときは、タブの名前を信じる
  return out.length ? out : byName;
}

/** 「105」のように決めきれないタブは、前後のタブの日付から決める */
function resolveMondays_(found) {
  found.forEach(function (w, i) {
    if (w.cands.length === 1) { w.monday = w.cands[0]; return; }
    var prev = '', next = '9999';
    for (var j = i - 1; j >= 0; j--) if (found[j].monday) { prev = found[j].monday; break; }
    for (var j2 = i + 1; j2 < found.length; j2++) if (found[j2].cands.length === 1) { next = found[j2].cands[0]; break; }
    var fit = w.cands.filter(function (c) { return c > prev && c < next; });
    w.monday = fit.length === 1 ? fit[0] : null;
  });
  return found.filter(function (w) { return w.monday; })
    .map(function (w) { return { name: w.name, monday: w.monday, ab: w.ab }; })
    .sort(function (a, b) { return a.monday < b.monday ? -1 : a.monday > b.monday ? 1 : 0; });
}

/** ひな形：同じA週/B週の、いちばん新しい週のタブ（なければ「A週」などのタブ） */
function pickTemplate_(ss, existing, ab) {
  for (var i = existing.length - 1; i >= 0; i--) {
    if (existing[i].ab === ab) { var s = ss.getSheetByName(existing[i].name); if (s) return s; }
  }
  // A週/B週のマスがない週案は、いちばん新しい週のタブ
  for (var j = existing.length - 1; j >= 0; j--) {
    if (!existing[j].ab) { var s2 = ss.getSheetByName(existing[j].name); if (s2) return s2; }
  }
  var ok = ss.getSheets().filter(function (s) { return Object.keys(shapeOf_(displayValues_(s)).classes).length; });
  var n = function (s) { return norm_(s.getName()).replace(/\s/g, ''); };
  return ok.filter(function (s) { return n(s) === ab + '週'; })[0] || ok.filter(function (s) { return n(s).indexOf(ab + '週') >= 0; })[0] || ok[0] || null;
}

function isWeekName_(name) { return /^\s*\d{1,2}(?:[\/／月]\d{1,2}日?|\d{1,3})\s*$/.test(weekKey_(name)); }
/** タブの名前の後ろの「～」をとる（1年生・2年生の週案は「825~」「4月13日～」の形） */
function weekKey_(name) { return norm_(name).replace(/[~〜～]+$/, '').trim(); }

/** 色のもと：日付でないひな形のタブ（「A週」→「A週（後期）」など→「原本」の順）。前の週のグレー（休みの日）をうつさないため */
function colorSource_(ss, ab) {

  var n = function (s) { return norm_(s.getName()).replace(/\s/g, ''); };

    ok.filter(function (s) { return n(s).indexOf(ab + '週') >= 0; })[0] ||
    ok.filter(function (s) { return n(s).indexOf('原本') >= 0; })[0] || null;
}

// ---------- 1週分を書きこむ ----------
/** 月〜金のマス（朝学〜宿題）を空にして、日付・A週/B週・教科を入れる。罫線・結合・集計はコピーのまま。
 *  色はひな形のタブから。祝日（振替休日・国民の休日も）はグレーにして、教科は入れない */
function fillWeek_(sh, week, grids, colorSrc, bunkatsu) {
  var vals = displayValues_(sh);
  var shape = shapeOf_(vals);
  var notes = [];
  if (shape.abCell) sh.getRange(shape.abCell.r, shape.abCell.c).setValue(week.ab + '週');   // A週/B週のマスがない週案なら書かない
  notes = notes.concat(writeDates_(sh, week, vals, shape));
  var srcShape = colorSrc ? shapeOf_(displayValues_(colorSrc)) : null;
  var holidays = {};
  Object.keys(shape.classes).forEach(function (cname) {
    var cl = shape.classes[cname];
    var grid = grids && grids[cname];   // grids が null：教科は入れない
    if (!grid && grids) notes.push(week.name + '：' + cname + 'の時間割がないので、教科は空です');
    var scl = srcShape && srcShape.classes[cname];
    var sameShape = scl && scl.firstRow === cl.firstRow && scl.lastRow === cl.lastRow && scl.rows['6'] === cl.rows['6'];
    DAYS.forEach(function (day, i) {
      var col = cl.dayCols[day];
      var date = addDays_(week.monday, i);
      var hol = holidayName_(date);
      if (hol) holidays[date] = hol;
      if (cl.firstRow && cl.lastRow >= cl.firstRow) {
        var n = cl.lastRow - cl.firstRow + 1;
        var blank = []; for (var k = 0; k < n; k++) blank.push(['']);
        sh.getRange(cl.firstRow, col, n, 1).setValues(blank);   // 前の週のメモを消す（書式は残る）
        // 色はひな形から（前の週の休みのグレーなどをうつさない）。グレーだけ残し、黄色などほかの色は色なしにする
        var bgs = (sameShape && scl.dayCols[day] ? colorSrc.getRange(cl.firstRow, scl.dayCols[day], n, 1) : sh.getRange(cl.firstRow, col, n, 1)).getBackgrounds();
        sh.getRange(cl.firstRow, col, n, 1).setBackgrounds(bgs.map(function (row) { return [isGray_(row[0]) ? row[0] : null]; }));
        if (hol) {   // 祝日：朝学〜6校時の下までグレー
          var end = cl.rows['6'] ? cl.rows['6'] + 2 : cl.lastRow;
          sh.getRange(cl.firstRow, col, end - cl.firstRow + 1, 1).setBackground(HOLIDAY_GRAY);
        }
      }
      if (!grid || hol) return;
      SLOTS.forEach(function (slot) {
        var r = cl.rows[slot];
        if (!r) return;
        var word = toShuanWord_(grid[day][slot], week.ab);
        sh.getRange(r, col).setValue(word);
        // 算数（1〜6校時）：教科の下のマスを黄色にして「分割」（パネルで選んだときだけ）
        if (bunkatsu && slot !== '朝' && word === '算数' && r + 1 <= cl.lastRow) sh.getRange(r + 1, col).setValue('分割').setBackground(BUNKATSU_YELLOW);
      });
    });
  });
  Object.keys(holidays).forEach(function (d) {
    var dd = dateOf_(d);
    notes.push(week.name + '：' + (dd.getUTCMonth() + 1) + '月' + dd.getUTCDate() + '日は' + holidays[d] + 'なのでグレーにしました');
  });
  return notes;
}

// ---------- 祝日（カレンダーどおり。外のサービスは使わず、法律の決まりで計算する） ----------
var HOLIDAY_GRAY = '#cccccc';
var BUNKATSU_YELLOW = '#ffff00';   // 算数の「分割」のマス（ほかのタブと同じ黄色）

/** グレー（白・黒はのぞく）なら true。'#rrggbb' の赤・緑・青がほぼ同じ */
function isGray_(hex) {
  var m = /^#?([0-9a-f]{2})([0-9a-f]{2})([0-9a-f]{2})$/i.exec(String(hex || ''));
  if (!m) return false;
  var v = [parseInt(m[1], 16), parseInt(m[2], 16), parseInt(m[3], 16)];
  var max = Math.max.apply(null, v), min = Math.min.apply(null, v);
  return max - min <= 12 && max < 250 && max > 20;
}

/** 'YYYY-MM-DD' が祝日なら名前、ちがえば '' */
function holidayName_(iso) {
  var y = Number(iso.slice(0, 4));
  return holidaysOf_(y)[iso] || '';
}

function holidaysOf_(y) {
  var h = {};
  var put = function (m, d, name) { h[y + '-' + pad_(m) + '-' + pad_(d)] = name; };
  var nthMonday = function (m, n) { var first = new Date(Date.UTC(y, m - 1, 1)).getUTCDay(); return 1 + ((8 - first) % 7) + (n - 1) * 7; };
  var shunbun = Math.floor(20.8431 + 0.242194 * (y - 1980) - Math.floor((y - 1980) / 4));
  var shubun = Math.floor(23.2488 + 0.242194 * (y - 1980) - Math.floor((y - 1980) / 4));
  put(1, 1, '元日'); put(1, nthMonday(1, 2), '成人の日'); put(2, 11, '建国記念の日'); put(2, 23, '天皇誕生日');
  put(3, shunbun, '春分の日'); put(4, 29, '昭和の日'); put(5, 3, '憲法記念日'); put(5, 4, 'みどりの日'); put(5, 5, 'こどもの日');
  put(7, nthMonday(7, 3), '海の日'); put(8, 11, '山の日'); put(9, nthMonday(9, 3), '敬老の日'); put(9, shubun, '秋分の日');
  put(10, nthMonday(10, 2), 'スポーツの日'); put(11, 3, '文化の日'); put(11, 23, '勤労感謝の日');
  var keys = Object.keys(h).sort();
  // 国民の休日：祝日と祝日にはさまれた日
  keys.forEach(function (k) {
    var mid = addDays_(k, 1), next = addDays_(k, 2);
    if (h[next] && !h[mid] && dateOf_(mid).getUTCDay() !== 0) h[mid] = '国民の休日';
  });
  // 振替休日：祝日が日曜なら、そのあとの祝日でない最初の日
  Object.keys(h).sort().forEach(function (k) {
    if (dateOf_(k).getUTCDay() !== 0 || h[k] === '振替休日') return;
    var d = addDays_(k, 1);
    while (h[d]) d = addDays_(d, 1);
    h[d] = '振替休日';
  });
  return h;
}

/** 月〜金の日付を書く。月も見えるように「1月4日」の形（下の行で、もう月と日の出る形ならその形のまま） */
function writeDates_(sh, week, vals, shape) {
  var notes = [];
  Object.keys(shape.classes).forEach(function (cname) {
    var cl = shape.classes[cname];
    var dr = dateRowOf_(vals, cl);
    if (!dr) { notes.push(week.name + '：' + cname + 'の日付の行が見つかりません'); return; }
    var fmt = dr.full ? String(sh.getRange(dr.row, cl.dayCols['月']).getNumberFormat() || '') : '';
    if (!/m/i.test(fmt) || !/d/i.test(fmt)) fmt = 'm"月"d"日"';
    DAYS.forEach(function (day, i) {
      var d = dateOf_(addDays_(week.monday, i));
      sh.getRange(dr.row, cl.dayCols[day]).setValue(new Date(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate())).setNumberFormat(fmt);
    });
  });
  return notes;
}

// ---------- 時間割を読む ----------
/** タブ区切りの文字（スプレッドシートのコピー）を表にする */
function parseTsv_(text) {
  var rows = [], row = [], cell = '', q = false;
  var s = String(text).replace(/\r\n?/g, '\n');
  for (var i = 0; i < s.length; i++) {
    var ch = s.charAt(i);
    if (q) {
      if (ch === '"' && s.charAt(i + 1) === '"') { cell += '"'; i++; } else if (ch === '"') q = false; else cell += ch;
    } else if (ch === '"' && cell === '') q = true;
    else if (ch === '\t') { row.push(cell); cell = ''; } else if (ch === '\n') { row.push(cell); rows.push(row); row = []; cell = ''; } else cell += ch;
  }
  if (cell !== '' || row.length) { row.push(cell); rows.push(row); }
  return rows;
}

/** 「月 火 水 木 金」の見出しの下の「1」〜「6」の行を校時、1校時より前の行を朝学とみる */
function parseTimetable_(rows) {
  var t = rows.map(function (r) { return r.map(norm_); });
  var h = -1, dayCols = null;
  for (var i = 0; i < t.length && h < 0; i++) {
    var cols = DAYS.map(function (d) { return t[i].indexOf(d); });
    var inOrder = cols.every(function (c, k) { return c >= 0 && (k === 0 || c > cols[k - 1]); });
    if (inOrder) { h = i; dayCols = cols; }
  }
  if (h < 0) return { ok: false, reason: '「月 火 水 木 金」の見出しが見つかりません。見出しの行もいっしょにコピーしてください' };
  var grid = {};
  DAYS.forEach(function (d) { grid[d] = {}; SLOTS.forEach(function (s) { grid[d][s] = ''; }); });
  var found = {};
  for (var j = h + 1; j < t.length; j++) {
    var r = t[j];
    var others = r.filter(function (_, c) { return dayCols.indexOf(c) < 0; });
    var vals = dayCols.map(function (c) { return r[c] || ''; });
    if (!vals.some(Boolean)) continue;
    var slot = null;
    others.forEach(function (v) { var m = /^([1-6])(校時|時間目)?$/.exec(v); if (m && !slot) slot = m[1]; });
    if (!slot && !Object.keys(found).length) slot = '朝';
    if (!slot || found[slot]) continue;
    found[slot] = true;
    DAYS.forEach(function (d, k) { grid[d][slot] = vals[k]; });
  }
  var missing = ['1', '2', '3', '4', '5', '6'].filter(function (s) { return !found[s]; });
  if (missing.length === 6) return { ok: false, reason: '1〜6校時の行が見つかりません。左の「1」「2」…の列もいっしょにコピーしてください' };
  return { ok: true, grid: grid, warnings: missing.length ? [missing.join('・') + '校時の行が見つかりませんでした'] : [] };
}

/** 「総/体」は A週なら「総合」、B週なら「体育」。「国(書写)」は「国語（書写）」 */
function toShuanWord_(raw, ab) {
  var s = norm_(raw);
  if (/[\/／]/.test(s)) { var p = s.split(/[\/／]/).map(norm_); if (p.length === 2) s = ab === 'B' ? p[1] : p[0]; }
  if (SHORT[s]) return SHORT[s];
  var m = /^(.)\((.+)\)$/.exec(s);
  if (m && SHORT[m[1]]) return SHORT[m[1]] + '（' + m[2] + '）';
  return s;
}

// ---------- 週の予定 ----------
/** from〜to の月曜日ごとに1週。A週/B週は前の週の反対から交互（abOverride で1週ずつ変えられる） */
function planWeeks_(from, to, existing, abOverride) {
  var out = [];
  if (!from || !to || from > to) return out;
  var start = mondayOf_(from);
  var before = existing.filter(function (w) { return w.monday < start; }).pop();
  var ab = before && before.ab ? (before.ab === 'A' ? 'B' : 'A') : 'A';
  for (var mon = start; mon <= to; mon = addDays_(mon, 7)) {
    if (abOverride[mon]) ab = abOverride[mon];
    var d = dateOf_(mon);
    out.push({
      monday: mon, name: (d.getUTCMonth() + 1) + '月' + d.getUTCDate() + '日', ab: ab,
      exists: existing.some(function (w) { return w.monday === mon; }),
    });
    ab = ab === 'A' ? 'B' : 'A';
  }
  return out;
}

function dateOf_(s) { var p = s.split('-').map(Number); return new Date(Date.UTC(p[0], p[1] - 1, p[2])); }
function addDays_(s, n) { return iso_(new Date(dateOf_(s).getTime() + n * 86400000)); }
function mondayOf_(s) { return addDays_(s, -((dateOf_(s).getUTCDay() + 6) % 7)); }
function schoolYear_() { var d = new Date(); return d.getMonth() + 1 >= 4 ? d.getFullYear() : d.getFullYear() - 1; }
function schoolYearOf_(iso) { if (!iso) return 0; var d = dateOf_(iso); return d.getUTCMonth() + 1 >= 4 ? d.getUTCFullYear() : d.getUTCFullYear() - 1; }
